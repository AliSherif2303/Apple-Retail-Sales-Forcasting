class Attention(Layer):

    def call(self, inputs):

        score = tf.nn.tanh(inputs)

        weights = tf.nn.softmax(score, axis=1)

        context = weights * inputs

        context = tf.reduce_sum(context, axis=1)

        return context

    X=[]

        X.append(data[i:i+seq_len])
